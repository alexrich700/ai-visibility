# SEO Audit — Implementation Spec for Replit

## Context for the AI Agent

You are building a new "SEO Audit" tab/experience within an existing Node.js + React application hosted on Replit. The app currently has a GEO (Generative Engine Optimization) AI Visibility tool that evaluates businesses against ChatGPT/Gemini/OpenAI. That stays as-is. This spec adds a comprehensive SEO audit pipeline that feeds a client-facing dashboard.

**Existing app capabilities to preserve:**
- GEO visibility engine (Gemini + OpenAI prompt generation and evaluation per service category)
- Magic-link sharing (token-based URLs for client read-only access)
- Supabase for data storage
- User authentication for specialists (backend admin)

**What's new:**
- Full SEO audit pipeline (10 dashboard sections)
- Google Ads Keyword Planner API integration (keyword research + PPC forecasting)
- DataForSEO API integration (SERP rankings, Maps geo grids, site crawl, backlinks)
- Google PageSpeed Insights API (Core Web Vitals)
- Google Places API (reviews, ratings)
- Interactive geo grid map (Leaflet.js)
- Present Mode (fullscreen slide-by-slide navigation)
- Specialist-facing onboarding wizard for audit configuration

---

## Tech Stack

- **Frontend:** React (existing) + Leaflet.js (new, for maps) + Recharts or Chart.js (new, for charts)
- **Backend:** Node.js (existing) + Express (existing)
- **Database:** Supabase (existing)
- **Auth:** Existing specialist auth + magic-link tokens for client view
- **APIs:** Google Ads API, DataForSEO REST API, Google PageSpeed Insights API, Google Places API, Anthropic API (Haiku + Sonnet), OpenAI API (existing), Gemini API (existing)

```javascript
/**
 * Run a full geo grid: multiple points × multiple keywords
 * Returns structured results for grid visualization
 */
async function runGeoGrid({
  centerLat, centerLng, gridSize, spacingMiles,
  keywords, clientBusinessName, competitorBusinessName
}) {
  // 1. Generate grid coordinates
  const points = generateGridPoints(centerLat, centerLng, gridSize, spacingMiles);
  // points = [{lat, lng, row, col}, ...] (e.g., 169 for 13×13)

  // 2. Build API tasks: points × keywords
  const tasks = [];
  for (const keyword of keywords) {
    for (const point of points) {
      tasks.push({
        keyword,
        location_coordinate: `${point.lat},${point.lng},17z`,
        language_code: 'en',
        depth: 20,
        tag: JSON.stringify({ keyword, row: point.row, col: point.col })
      });
    }
  }

  // 3. Batch into groups of 100 and POST
  // /serp/google/maps/task_post (standard queue, cheapest)
  // OR /serp/google/maps/live/advanced (real-time, 2x cost)
  const batches = chunk(tasks, 100);
  const taskIds = [];
  for (const batch of batches) {
    const response = await post(`${BASE_URL}/serp/google/maps/task_post`, batch);
    taskIds.push(...response.tasks.map(t => t.id));
  }

  // 4. Poll for results (standard queue takes 5-30 seconds)
  // GET /serp/google/maps/task_get/{task_id}
  // Collect all results

  // 5. Parse results into grid structure
  // For each point, find client rank and competitor rank
  const gridResults = {};
  for (const result of allResults) {
    const { keyword, row, col } = JSON.parse(result.tag);
    const clientRank = findRank(result.items, clientBusinessName);
    const competitorRank = findRank(result.items, competitorBusinessName);

    if (!gridResults[keyword]) gridResults[keyword] = [];
    gridResults[keyword].push({
      row, col,
      lat: result.location_coordinate.split(',')[0],
      lng: result.location_coordinate.split(',')[1],
      clientRank,     // null if not found in top 20
      competitorRank, // null if not found in top 20
      topResults: result.items.slice(0, 5).map(i => ({
        rank: i.rank_absolute,
        name: i.title,
        rating: i.rating?.value,
        reviews: i.rating?.votes_count,
        domain: i.domain
      }))
    });
  }

  // 6. Calculate metrics per keyword
  const metrics = {};
  for (const [keyword, points] of Object.entries(gridResults)) {
    const total = points.length;
    const clientTop3 = points.filter(p => p.clientRank && p.clientRank <= 3).length;
    const clientRanked = points.filter(p => p.clientRank != null);
    const compTop3 = points.filter(p => p.competitorRank && p.competitorRank <= 3).length;
    const compRanked = points.filter(p => p.competitorRank != null);

    metrics[keyword] = {
      clientSoLV: ((clientTop3 / total) * 100).toFixed(1),
      clientAvgRank: clientRanked.length > 0
        ? (clientRanked.reduce((s, p) => s + p.clientRank, 0) / clientRanked.length).toFixed(1)
        : null,
      competitorSoLV: ((compTop3 / total) * 100).toFixed(1),
      competitorAvgRank: compRanked.length > 0
        ? (compRanked.reduce((s, p) => s + p.competitorRank, 0) / compRanked.length).toFixed(1)
        : null,
      totalPoints: total,
      clientVisiblePoints: clientRanked.length,
      clientTop3Points: clientTop3
    };
  }

  return { gridResults, metrics };
}

/**
 * Generate lat/lng grid points centered on a location
 */
function generateGridPoints(centerLat, centerLng, gridSize, spacingMiles) {
  const points = [];
  const halfGrid = Math.floor(gridSize / 2);

  // Degrees per mile varies by latitude
  const latDegreesPerMile = 1 / 69.0; // ~69 miles per degree latitude
  const lngDegreesPerMile = 1 / (69.0 * Math.cos(centerLat * Math.PI / 180));

  for (let row = -halfGrid; row <= halfGrid; row++) {
    for (let col = -halfGrid; col <= halfGrid; col++) {
      points.push({
        lat: parseFloat((centerLat + (row * spacingMiles * latDegreesPerMile)).toFixed(7)),
        lng: parseFloat((centerLng + (col * spacingMiles * lngDegreesPerMile)).toFixed(7)),
        row: row + halfGrid, // 0-indexed
        col: col + halfGrid
      });
    }
  }
  return points;
}

function findRank(items, businessName) {
  if (!items || !businessName) return null;
  const normalized = businessName.toLowerCase().trim();
  const match = items.find(item =>
    item.title && item.title.toLowerCase().includes(normalized)
  );
  return match ? match.rank_absolute : null;
}

/**
```

```javascript
 * Returns findings as pass/fail/warning

function runTechnicalChecks(psiData, crawlData) {
  const findings = [];

  // ── Speed & Performance ──
  const lcp = psiData.lighthouseResult.audits['largest-contentful-paint'].numericValue;
  findings.push({
    category: 'speed',
    check_name: 'Largest Contentful Paint (LCP)',
    status: lcp <= 2500 ? 'pass' : lcp <= 4000 ? 'warning' : 'fail',
    value: `${(lcp / 1000).toFixed(1)}s`,
    threshold: '≤ 2.5s',
    impact: lcp > 4000 ? 'high' : 'medium'
  });

  const inp = psiData.lighthouseResult.audits['interaction-to-next-paint']?.numericValue;
  findings.push({
    category: 'speed',
    check_name: 'Interaction to Next Paint (INP)',
    status: inp <= 200 ? 'pass' : inp <= 500 ? 'warning' : 'fail',
    value: `${inp}ms`,
    threshold: '≤ 200ms',
    impact: inp > 500 ? 'high' : 'medium'
  });

  const cls = psiData.lighthouseResult.audits['cumulative-layout-shift'].numericValue;
  findings.push({
    category: 'speed',
    check_name: 'Cumulative Layout Shift (CLS)',
    status: cls <= 0.1 ? 'pass' : cls <= 0.25 ? 'warning' : 'fail',
    value: cls.toFixed(3),
    threshold: '≤ 0.1',
    impact: cls > 0.25 ? 'high' : 'medium'
  });

  const perfScore = psiData.lighthouseResult.categories.performance.score * 100;
  findings.push({
    category: 'speed',
    check_name: 'Overall Performance Score',
    status: perfScore >= 90 ? 'pass' : perfScore >= 50 ? 'warning' : 'fail',
    value: `${perfScore}/100`,
    threshold: '≥ 90',
    impact: perfScore < 50 ? 'high' : 'medium'
  });

  // ── Security ──
  findings.push({
    category: 'security',
    check_name: 'HTTPS',
    status: psiData.lighthouseResult.audits['is-on-https']?.score === 1 ? 'pass' : 'fail',
    value: psiData.lighthouseResult.audits['is-on-https']?.score === 1 ? 'Yes' : 'No',
    threshold: 'Required',
    impact: 'high'

  if (crawlData) {

      status: crawlData.has_sitemap ? 'pass' : 'fail',
      value: crawlData.has_sitemap ? 'Yes' : 'No',

    findings.push({

      status: crawlData.broken_links_count === 0 ? 'pass'
             : crawlData.broken_links_count <= 5 ? 'warning' : 'fail',
      value: `${crawlData.broken_links_count} broken links`,
      threshold: '0',
      impact: crawlData.broken_links_count > 10 ? 'high' : 'medium'
    });

    findings.push({
      category: 'crawlability',
      check_name: 'Redirect Chains',
      status: crawlData.redirect_count <= 3 ? 'pass' : 'warning',
      value: `${crawlData.redirect_count} redirects`,
      threshold: '≤ 3',
      impact: 'medium'
    });

    findings.push({
      category: 'crawlability',
      check_name: 'Non-Indexable Pages',
      status: crawlData.non_indexable_count === 0 ? 'pass' : 'warning',
      value: `${crawlData.non_indexable_count}`,
      threshold: '0 unexpected',
      impact: 'medium'

    // ── On-Page SEO ──
    findings.push({

/**
 * Calculate letter grade from findings
 */
function calculateSiteHealthGrade(findings) {
  const total = findings.length;
  const passed = findings.filter(f => f.status === 'pass').length;
  const ratio = passed / total;

  if (ratio >= 0.9) return 'A';
  if (ratio >= 0.8) return 'B';
  if (ratio >= 0.65) return 'C';
  if (ratio >= 0.5) return 'D';
  return 'F';
}
```

```javascript
// server/pipeline/orchestrator.js

const { EventEmitter } = require('events');
const keywordPlanner = require('../services/googleAdsKeywordPlanner');
const d4seo = require('../services/dataForSeo');
const pagespeed = require('../services/pagespeed');
const llm = require('../services/llm');
const db = require('../services/supabase');

class AuditPipeline extends EventEmitter {
  constructor(auditId) {
    super();
    this.auditId = auditId;
    this.audit = null;
  }

  async run() {
    this.audit = await db.getAudit(this.auditId);

    const stages = [
      { name: 'intake',           fn: () => this.stageIntake() },
      { name: 'technical_audit',  fn: () => this.stageTechnicalAudit() },
      { name: 'keyword_research', fn: () => this.stageKeywordResearch() },
      { name: 'serp_rankings',    fn: () => this.stageSerpRankings() },
      { name: 'geo_grid',         fn: () => this.stageGeoGrid(),         skip: this.audit.business_type === 'national' },
      { name: 'competitive_intel',fn: () => this.stageCompetitiveIntel() },
      { name: 'geo_visibility',   fn: () => this.stageGeoVisibility() },
      { name: 'review_health',    fn: () => this.stageReviewHealth() },
      { name: 'gap_analysis',     fn: () => this.stageGapAnalysis() },
      { name: 'content_scoring',  fn: () => this.stageContentScoring() },
      { name: 'scoping',          fn: () => this.stageScoping() },
      { name: 'report',           fn: () => this.stageReport() },
    ];

    for (const stage of stages) {
      if (stage.skip) {
        this.emit('stage_skip', stage.name);
        continue;
      }

      await db.logStageStart(this.auditId, stage.name);
      this.emit('stage_start', stage.name);

      try {
        const result = await stage.fn();
        await db.logStageComplete(this.auditId, stage.name, result);
        this.emit('stage_complete', stage.name, result);
      } catch (error) {
        await db.logStageError(this.auditId, stage.name, error.message);
        this.emit('stage_error', stage.name, error);
        // Continue with degraded data — don't halt pipeline
      }

      await db.updateAudit(this.auditId, { current_stage: stage.name });
    }

    await db.updateAudit(this.auditId, { status: 'completed' });
    this.emit('pipeline_complete');
  }

  // ── Stage 1: Intake ──────────────────────────────────────
  async stageIntake() {
    // 1. Start site crawl (DataForSEO)
    const crawlTaskId = await d4seo.startSiteCrawl({
      targetUrl: this.audit.business_url,
      maxPages: 500
    });

    // 2. Geocode business address → lat/lng (if not already set)
    if (!this.audit.business_lat) {
      const coords = await geocodeAddress(this.audit.business_address);
      await db.updateAudit(this.auditId, {
        business_lat: coords.lat,
        business_lng: coords.lng
      });
      this.audit.business_lat = coords.lat;
      this.audit.business_lng = coords.lng;
    }

    // 3. Wait for crawl to complete (poll every 5 seconds)
    let crawlComplete = false;
    while (!crawlComplete) {
      const summary = await d4seo.getCrawlSummary(crawlTaskId);
      if (summary.crawl_progress === 'finished') {
        crawlComplete = true;
        await db.updateAudit(this.auditId, { crawl_task_id: crawlTaskId });
      } else {
        await sleep(5000);
      }
    }

    // 4. Extract page hierarchy from crawl
    const pages = await d4seo.getCrawlPages(crawlTaskId, { status_code: 200 });

    // 5. If services not yet classified, use Haiku to extract from page titles/content
    if (this.audit.services.length === 0) {
      const extractedServices = await llm.extractServicesFromPages(pages);
      await db.updateAudit(this.auditId, { services: extractedServices });
      this.audit.services = extractedServices;
    }

    return { crawlTaskId, pagesFound: pages.length, services: this.audit.services.length };
  }

  // ── Stage 2: Technical Audit ──────────────────────────────
  async stageTechnicalAudit() {
    // 1. Run PageSpeed Insights (mobile + desktop)
    const [mobileData, desktopData] = await Promise.all([
      pagespeed.getPageSpeedInsights(this.audit.business_url, 'mobile'),
      pagespeed.getPageSpeedInsights(this.audit.business_url, 'desktop')
    ]);

    // 2. Get crawl data
    const crawlSummary = await d4seo.getCrawlSummary(this.audit.crawl_task_id);

    // 3. Run 37 binary checks
    const findings = pagespeed.runTechnicalChecks(mobileData, crawlSummary);

    // 4. Calculate grade
    const grade = pagespeed.calculateSiteHealthGrade(findings);

    // 5. Save to DB
    await db.bulkInsertTechnicalFindings(this.auditId, findings);
    await db.updateAudit(this.auditId, { site_health_grade: grade });

    return {
      grade,
      total: findings.length,
      pass: findings.filter(f => f.status === 'pass').length,
      fail: findings.filter(f => f.status === 'fail').length
    };
  }

  // ── Stage 9: Gap Analysis ──────────────────────────────────
  async stageGapAnalysis() {
    const keywords = await db.getAuditKeywords(this.auditId);
    const { services, service_area_cities } = this.audit;

    const gaps = [];

    // Service page gaps: keyword has no matching service page
    for (const service of services) {
      const serviceKws = keywords.filter(k =>
        k.target_service === service.name && k.page_type === 'service_page'
      );
      for (const kw of serviceKws) {
        if (!kw.existing_page_url) {
          gaps.push({
            gap_type: 'service_page',
            target_keyword: kw.keyword,
            target_service: service.name,
            search_volume: kw.search_volume,
            priority: kw.search_volume > 500 ? 'high' : 'medium',
            estimated_hours: 3, // service page creation
            status: 'missing'
          });
        }
      }
    }

    // City page gaps
    for (const city of service_area_cities) {
      const cityKws = keywords.filter(k =>
        k.target_city === city.name && k.page_type === 'city_page'
      );
      for (const kw of cityKws) {
        if (!kw.existing_page_url) {
          gaps.push({
            gap_type: 'city_page',
            target_keyword: kw.keyword,
            target_city: city.name,
            search_volume: kw.search_volume,
            priority: 'high',
            estimated_hours: 3, // city page template + localization
            status: 'missing'
          });
        }
      }
    }

    // Sub-city page gaps (service × city combos)
    for (const city of service_area_cities) {
      for (const service of services) {
        const subCityKws = keywords.filter(k =>
          k.target_city === city.name &&
          k.target_service === service.name &&
          k.page_type === 'subcity_page'
        );
        for (const kw of subCityKws) {
          if (!kw.existing_page_url) {
            gaps.push({
              gap_type: 'subcity_page',
              target_keyword: kw.keyword,
              target_city: city.name,
              target_service: service.name,
              search_volume: kw.search_volume,
              priority: 'medium',
              estimated_hours: 0.5, // bulk creation from template
              status: 'missing'
            });
          }
        }
      }
    }

    // Comparison page opportunities
    // "X vs Y" for top competitors in each service area
    for (const comp of this.audit.competitors) {
      gaps.push({
        gap_type: 'comparison',
        target_keyword: `${this.audit.business_name} vs ${comp.name}`,
        search_volume: null, // estimated
        priority: 'medium',
        estimated_hours: 2,
        status: 'missing'
      });
    }

    await db.bulkInsertContentGaps(this.auditId, gaps);
    await db.updateAudit(this.auditId, {
      total_content_gaps: gaps.length,
      total_keyword_gaps: keywords.filter(k => !k.existing_page_url).length
    });

    return { totalGaps: gaps.length };
  }

  // ── Stage 10: Content Scoring ──────────────────────────────
  async stageContentScoring() {
    const keywords = await db.getAuditKeywords(this.auditId);
    const withPages = keywords.filter(k => k.existing_page_url);
    const uniquePages = [...new Set(withPages.map(k => k.existing_page_url))];

    for (const pageUrl of uniquePages) {
      const pageKeyword = withPages.find(k => k.existing_page_url === pageUrl);
      // Fetch page content (from crawl data or direct fetch)
      const pageContent = await fetchPageContent(pageUrl);

      const score = await llm.scorePage(
        pageUrl, pageContent,
        pageKeyword.keyword, pageKeyword.intent
      );

      // If score indicates optimization needed, add to content gaps
      if (score.action === 'optimize' || score.action === 'rewrite') {
        await db.insertContentGap(this.auditId, {
          gap_type: 'needs_optimization',
          target_keyword: pageKeyword.keyword,
          estimated_hours: score.estimated_optimization_hours,
          status: 'needs_optimization'
        });
      }
    }

    return { pagesScored: uniquePages.length };
  }

  // ── Stage 11: Scoping Engine ───────────────────────────────
  async stageScoping() {
    const technicalFindings = await db.getTechnicalFindings(this.auditId);
    const contentGaps = await db.getContentGaps(this.auditId);
    const failedChecks = technicalFindings.filter(f => f.status === 'fail');

    const deliverables = [];
    let sortOrder = 0;

    // Phase 1: Technical fixes + quick wins (Month 1-3)
    for (const finding of failedChecks) {
      deliverables.push({
        phase: 1,
        category: 'technical',
        title: `Fix: ${finding.check_name}`,
        description: finding.description,
        estimated_hours: finding.impact === 'high' ? 2 : 1,
        priority: finding.impact,
        sort_order: sortOrder++
      });
    }

    // Phase 1: Schema markup
    deliverables.push({
      phase: 1,
      category: 'technical',
      title: 'Implement LocalBusiness + Service schema across all pages',
      estimated_hours: 4,
      priority: 'high',
      sort_order: sortOrder++
    });

    // Phase 1: GBP optimization
    deliverables.push({
      phase: 1,
      category: 'gbp',
      title: 'Google Business Profile full optimization',
      estimated_hours: 3,
      priority: 'high',
      sort_order: sortOrder++
    });

    // Phase 2: Content build-out (Month 4-6)
    const servicePageGaps = contentGaps.filter(g => g.gap_type === 'service_page');
    if (servicePageGaps.length > 0) {
      deliverables.push({
        phase: 2,
        category: 'content',
        title: `Create ${servicePageGaps.length} service pages`,
        description: servicePageGaps.map(g => g.target_keyword).join(', '),
        estimated_hours: servicePageGaps.length * 3,
        priority: 'high',
        sort_order: sortOrder++
      });
    }

    const cityPageGaps = contentGaps.filter(g => g.gap_type === 'city_page');
    if (cityPageGaps.length > 0) {
      deliverables.push({
        phase: 2,
        category: 'geo_pages',
        title: `Create ${cityPageGaps.length} city pages`,
        description: cityPageGaps.map(g => g.target_city).join(', '),
        estimated_hours: cityPageGaps.length * 3, // 3 hrs for template, 0.5 for subsequent
        priority: 'high',
        sort_order: sortOrder++
      });
    }

    // Phase 3: Scale (Month 7-12)
    const subCityGaps = contentGaps.filter(g => g.gap_type === 'subcity_page');
    if (subCityGaps.length > 0) {
      deliverables.push({
        phase: 3,
        category: 'geo_pages',
        title: `Create ${subCityGaps.length} sub-city pages (service × city)`,
        estimated_hours: subCityGaps.length * 0.5,
        priority: 'medium',
        sort_order: sortOrder++
      });
    }

    const comparisonGaps = contentGaps.filter(g => g.gap_type === 'comparison');
    if (comparisonGaps.length > 0) {
      deliverables.push({
        phase: 3,
        category: 'content',
        title: `Create ${comparisonGaps.length} comparison pages`,
        estimated_hours: comparisonGaps.length * 2,
        priority: 'medium',
        sort_order: sortOrder++
      });
    }

    // Link building
    deliverables.push({
      phase: 3,
      category: 'backlinks',
      title: 'Monthly link building campaign',
      description: 'Local directories, guest posts, press releases',
      estimated_hours: 10, // per month
      priority: 'medium',
      sort_order: sortOrder++
    });

    await db.bulkInsertDeliverables(this.auditId, deliverables);

    const totalHours = deliverables.reduce((sum, d) => sum + d.estimated_hours, 0);
    await db.updateAudit(this.auditId, {
      total_deliverables: deliverables.length,
      estimated_total_hours: totalHours
    });

    return { deliverables: deliverables.length, totalHours };
  }

  // ── Stage 12: Report Generation ────────────────────────────
  async stageReport() {
    // Gather all data for executive narrative
    const auditSummary = await db.getFullAuditSummary(this.auditId);

    // Generate narrative via Sonnet
    const narrative = await llm.generateActionPlanNarrative(auditSummary);

    await db.updateAudit(this.auditId, {
      executive_narrative: narrative,
      status: 'completed'
    });

    // Generate magic link if not already set
    if (!this.audit.magic_link_token) {
      const token = generateSecureToken();
      await db.updateAudit(this.auditId, { magic_link_token: token });
    }

    return { completed: true };
  }
}

module.exports = { AuditPipeline };
```
